import { afterEach, expect, it, vi } from "vitest";
import { createPracticeApi } from "./api";
import type { MobileAuthController } from "../auth/mobile-auth";
import type { MobileReview } from "../lib/api";
const identity = "a".repeat(64);
const review: MobileReview = {
  takeId: "take", scriptId: "script", favorite: false, displayName: null,
  createdAt: "2026-09-22T00:00:00Z", reviewedAt: null, transcriptText: "Hello",
  audioIdentity: identity,
  evaluation: { score: 80, accuracyScore: 80, fluencyScore: 80, rhythmScore: 80, summaryJa: "良い", strengthsJa: [], weakWords: [], scriptWordCount: 1, transcriptWordCount: 1 },
  coach: { titleJa: "次", summaryJa: "次", bulletPointsJa: [], nextStepJa: "次", focusWords: [] }
};

function fixture() {
 const script={id:"script",title:"Canonical",content:"Hello",targetSeconds:60,locale:"en-US",createdAt:"2026-09-22",updatedAt:"2026-09-22"};
 let status=200, version: unknown=identity, block: ((path:string)=>Promise<void>)|undefined;
 const fetchImpl=vi.fn(async(input:string|URL|Request,init?:RequestInit)=>{
  const path=new URL(String(input)).pathname, capturedStatus=status, capturedVersion=version;
  const takeId=path.split("/").at(-1)!;
  const data=path.includes("/reviews/")?{review:{...review,takeId,audioIdentity:capturedVersion}}:path.endsWith("/scripts")?{scripts:[script]}:{script};
  await block?.(path);
  if(init?.method && init.method!=="GET")return new Response("{}",{status:503,headers:{"content-type":"application/json"}});
  if(path.endsWith("/audio"))return new Response(new Blob(["owned"],{type:"audio/wav"}),{headers:{"content-type":"audio/wav","content-disposition":"attachment; filename*=UTF-8''Take.wav","x-take-audio-identity":String(capturedVersion)}});
  return new Response(JSON.stringify(capturedStatus===200?{ok:true,data}:{ok:false,error:{code:"review_not_found"}}),{status:capturedStatus,headers:{"content-type":"application/json"}});
 });vi.stubGlobal("fetch",fetchImpl);
 const auth={getState:()=>({kind:"authenticated",userId:"owner"}),getAccessToken:async()=>"token"} as unknown as MobileAuthController;
 const api=createPracticeApi({auth,bffBaseUrl:"https://fixture.test",ownerUserId:"owner",onTiming:()=>undefined});
 const fresh=()=>{const s=api.reviewMemory!.getSnapshot("script/take");if(s.kind!=="ready")throw Error(s.kind);return s.data.review;};
 const stop=()=>{api.scriptsMemory!.revoke();api.reviewMemory!.revoke();api.savedTakeAudioMemory!.revoke();api.progressMemory!.revoke();};
 return{api,fetchImpl,fresh,stop,setStatus:(v:number)=>{status=v;},setVersion:(v:unknown)=>{version=v;},hold:(v:typeof block)=>{block=v;}};
}
afterEach(()=>vi.unstubAllGlobals());
it("cached Review never carries an audio capability; fresh revalidation preserves existing audio reuse and fresh Share",async()=>{
 const f=fixture(),m=f.api.reviewMemory!,leave=m.enter("script/take");await m.revalidate();const first=f.fresh();await f.api.prepareSavedTakeAudio!(first);leave();
 const cached=m.getSnapshot("script/take");expect(cached).not.toHaveProperty("data.review.audioVisit");expect(cached).toMatchObject({refreshing:true});
 const secondLeave=m.enter("script/take");await m.revalidate();const next=f.fresh();expect(next.audioVisit).not.toBe(first.audioVisit);await f.api.prepareSavedTakeAudio!(next);
 expect(f.fetchImpl.mock.calls.filter(([u])=>String(u).endsWith("/audio"))).toHaveLength(1);
 await f.api.downloadTakeAudio("take");expect(f.fetchImpl.mock.calls.filter(([u])=>String(u).endsWith("/audio"))).toHaveLength(2);secondLeave();f.stop();
});
it("late old route failure cannot invalidate a newer Review proof or reusable snapshot",async()=>{
 const f=fixture();let release!:()=>void;f.hold(path=>path.endsWith("/old")?new Promise<void>(r=>{release=r;}):Promise.resolve());
 const m=f.api.reviewMemory!;m.enter("script/old");const pending=m.revalidate();await vi.waitFor(()=>expect(release).toBeTypeOf("function"));m.enter("script/take");await m.revalidate();const current=f.fresh();await f.api.prefetchSavedTakeAudio!(current);
 release();await pending;expect(m.getSnapshot("script/take")).toMatchObject({refreshing:false});expect((await f.api.prepareSavedTakeAudio!(current)).kind).toBe("success");expect(f.fetchImpl.mock.calls.filter(([u])=>String(u).endsWith("/audio"))).toHaveLength(1);f.stop();
});
it.each([null,"invalid",{},0])("invalid identity %j never seeds reusable metadata",async version=>{
 const f=fixture();f.setVersion(version);const m=f.api.reviewMemory!,leave=m.enter("script/take");await m.revalidate();expect(m.getSnapshot("script/take").kind).toBe("ready");leave();expect(m.getSnapshot("script/take").kind).toBe("loading");f.stop();
});
it.each([401,403,404,503])("fresh %s removes cached metadata and audio authorization",async status=>{
 const f=fixture(),m=f.api.reviewMemory!,leave=m.enter("script/take");await m.revalidate();const old=f.fresh();await f.api.prepareSavedTakeAudio!(old);leave();f.setStatus(status);m.enter("script/take");await m.revalidate();expect(m.getSnapshot("script/take").kind).toBe("error");expect((await f.api.prepareSavedTakeAudio!(old)).kind).not.toBe("success");f.stop();
});
it("audio fetch/version failure clears reusable metadata but keeps current canonical Review readable",async()=>{
 const f=fixture(),m=f.api.reviewMemory!,leave=m.enter("script/take");await m.revalidate();const current=f.fresh();f.setVersion("b".repeat(64));expect((await f.api.prepareSavedTakeAudio!(current)).kind).toBe("invalid-response");expect(m.getSnapshot("script/take")).toMatchObject({kind:"ready",refreshing:false});leave();expect(m.getSnapshot("script/take").kind).toBe("loading");f.stop();
});
it.each(["favorite","name","create","evaluate","account-delete","voice-delete","voice-advance"])("actual %s mutation invalidates both snapshots even when the write fails",async mutation=>{
 const f=fixture(),scripts=f.api.scriptsMemory!,reviews=f.api.reviewMemory!;const a=scripts.enter("scripts"),b=reviews.enter("script/take");await Promise.all([scripts.revalidate(),reviews.revalidate()]);a();b();
 const request=mutation==="favorite"?f.api.updateTakeMetadata("take",{favorite:true}):mutation==="name"?f.api.updateTakeMetadata("take",{displayName:"New"}):mutation==="create"?f.api.createScript({title:"New",content:"Hello"}):mutation==="evaluate"?f.api.evaluateRecording({scriptId:"script",takeId:"take",recordingRef:"aaaaaaaa-aaaa-4aaa-8aaa-aaaaaaaaaaaa"}):mutation==="account-delete"?f.api.createAccountDeletionRequest():mutation==="voice-delete"?f.api.createVoiceDeletionRequest():f.api.advanceVoiceDeletion();
 expect(scripts.getSnapshot("scripts").kind).toBe("loading");expect(reviews.getSnapshot("script/take").kind).toBe("loading");await request;expect(scripts.getSnapshot("scripts").kind).toBe("loading");expect(reviews.getSnapshot("script/take").kind).toBe("loading");f.stop();
});
